Bluetooth Base Registry Settings

The following code example shows the Bluetooth base registry settings.

; @CESYSGEN IF BTD_TDI 
[HKEY_LOCAL_MACHINE\Comm\AFD]
    "Stacks"="irdastk,tcpstk,btd"
; @CESYSGEN ENDIF

; @CESYSGEN IF BTD_RFCOMM
[HKEY_LOCAL_MACHINE\ExtModems\bluetooth_dcn]
    "port"="COM8:"
    "DeviceType"=dword:0
    "FriendlyName"="BluetoothDCN"

[HKEY_LOCAL_MACHINE\ExtModems\bluetooth_dun]
    "port"="COM8:"
    "DeviceType"=dword:1
    "FriendlyName"="BluetoothDUN"

; @CESYSGEN ENDIF
; @CESYSGEN IF BTD_UART
;
;       Flags (ored):
;               1 - do not perform reset
;               2 - local name is not supported
;               4 - role switch on connection requests is not supported
;
[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\HCI]
    "Name"="COM1:"
    "flags"=dword:4
IF IMGERICSSON
    "baud"=dword:e100
ENDIF
IF IMGSIW
    "baud"=dword:1c200
    "resetdelay"=dword:1388
ENDIF

; @CESYSGEN ENDIF
