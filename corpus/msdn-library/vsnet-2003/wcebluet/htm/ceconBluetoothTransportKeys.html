Bluetooth Transport Settings

The following code example shows the Bluetooth universal transport manager registry settings.

; BthUniv first looks for PCMCIA cards. If none are present, it checks for builtin devices 
; in the order listed.
;
; All values from found key and respective subkey of devices\active are copied over to 
; [HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\HCI] before opening a connection to the device.

Nokia

; @CESYSGEN IF CE_MODULES_BTHNOKIA
[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\PCMCIA\Socket-CF+_Personal_Network_Card-6C76]
    "driver"="bthnokia.dll"
    "baud"=dword:2580
    "flags"=dword:3
; @CESYSGEN ENDIF

CSR

; @CESYSGEN IF CE_MODULES_BTHCSR
[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\PCMCIA\Socket-CF+_Serial_Port_Card_Rev_2.3-EF12]
    "driver"="bthcsr.dll"
    "baud"=dword:9600
    "flags"=dword:2

[HKEY_LOCAL_MACHINE\Drivers\PCMCIA\Brain_Boxes-Bluetooth_PC_Card-59C7]
    "DeviceArrayIndex"=dword:1
    "Prefix"="COM"
    "Dll"="Serial.Dll"
    "ResetDelay"=dword:300
    "NoIntrPend"=dword:1

[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\PCMCIA\Brain_Boxes-Bluetooth_PC_Card-59C7]
    "driver"="bthcsr.dll"
    "baud"=dword:1c200
    "flags"=dword:1
; @CESYSGEN ENDIF

UART

; @CESYSGEN IF CE_MODULES_BTHUART
[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\PCMCIA\SHARP-BLUETOOTH_Card-97DB]
    "flags"=dword:4
    "driver"="bthuart.dll"
    "baud"=dword:1c200
    "resetdelay"=dword:1388

[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\PCMCIA\Xircom-Bluetooth_Adapter-3618]
    "flags"=dword:4
    "driver"="bthuart.dll"
    "baud"=dword:1c20
; @CESYSGEN ENDIF
; @CESYSGEN IF CE_MODULES_BTHUART
[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\BuiltIn\2]
    "driver"="bthuart.dll"
    "flags"=dword:4
;      Ericsson dongle/dev board
IF IMGERICSSON
    "name"="COM1:"
    "baud"=dword:e100
    "resetdelay"=dword:0
ENDIF
;     SiW boxes
IF IMGSIW
    "name"="COM1:"
    "baud"=dword:1c200
    "resetdelay"=dword:1388
ENDIF
; @CESYSGEN ENDIF

USB

; @CESYSGEN IF CE_MODULES_BTHUSB
[HKEY_LOCAL_MACHINE\Software\Microsoft\Bluetooth\Transports\BuiltIn\1]
    "flags"=dword:4
    "driver"="bthusb.dll"
    "resetdelay"=dword:0
; @CESYSGEN ENDIF
