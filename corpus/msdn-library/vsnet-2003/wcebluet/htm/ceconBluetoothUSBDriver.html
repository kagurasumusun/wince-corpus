Bluetooth USB Driver Settings

The following code example shows the Bluetooth USB driver settings.

; @CESYSGEN IF CE_MODULES_WENDYSER
[HKEY_LOCAL_MACHINE\Drivers\PCMCIA\Socket-CF+_Personal_Network_Card-6C76]
    "DeviceArrayIndex"=dword:00000000
    "DeviceType"=dword:00000002
    "Prefix"="COM"
    "DLL"="wendyser.dll"
; @CESYSGEN ENDIF
; @CESYSGEN IF CE_MODULES_BTHUNIV
[HKEY_LOCAL_MACHINE\Drivers\USB\LoadClients\Default\224_1\Default\Bluetooth_Universal_Driver]
    "DLL"="bthuniv.dll"
[HKEY_LOCAL_MACHINE\Drivers\USB\LoadClients\Default\Default\224_1\Bluetooth_Universal_Driver]
    "DLL"="bthuniv.dll"
; @CESYSGEN ENDIF
; @CESYSGEN IF CE_MODULES_BTHUSB
[HKEY_LOCAL_MACHINE\Drivers\USB\LoadClients\Default\224_1_1\Default\Bluetooth_USB_Driver]
    "DLL"="bthusb.dll"
[HKEY_LOCAL_MACHINE\Drivers\USB\LoadClients\Default\Default\224_1_1\Bluetooth_USB_Driver]
    "DLL"="bthusb.dll"
; @CESYSGEN ENDIF
