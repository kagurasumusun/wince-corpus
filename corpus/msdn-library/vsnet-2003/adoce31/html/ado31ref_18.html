Append (Tables)

This method adds a new Table object to the Tables collection.

Syntax

Tables.Append Table

Parameters

Parameter Description
Table A Variant value containing a reference to the Table to append or the name of the table to create and append

Return Values

None

Remarks

An error will occur if the provider does not support creating tables.

Example

The following code demonstrates how to use Append (Table).

Dim m_cat, m_col, tbl, col, strConn, stTableName, stColumnName, stExp, stAct

    Set m_cat = CreateObject("adoxce.catalog.3.1")
    Set m_col = CreateObject("adoxce.column.3.1")
    Set tbl = CreateObject("adoxce.table.3.1")
    Set col = CreateObject("adoxce.column.3.1")

    'strConn = "provider=microsoft.sqlserver.oledb.ce.1.0;data source=pubs.db;"
    strConn = "provider=microsoft.jet.oledb.4.0;data source=temp.mdb;"
    m_cat.ActiveConnection = strConn

    stTableName = "ColAppScen18"
    If adox_ExistsTable(m_cat, stTableName)

m_cat.Tables.Delete stTableName
    End If

    tbl.Name = stTableName

    stColumnName = "Column1"
    m_col.Name = stColumnName
    m_col.Type = adSmallInt

    tbl.Columns.Append m_col
    m_cat.Tables.Append tbl

    'create a new column
    col.Name = "Column2"
    col.Type = adSmallInt

    'Set parentcatalog
    col.ParentCatalog = m_cat

    col.Properties.Refresh

    m_cat.Tables(stTableName).Columns.Append col

    stExp = "Column2"
    stAct = m_cat.Tables(stTableName).Columns(1).Name

    'cleanup
    m_cat.Tables.Delete stTableName
    Set tbl = Nothing
    Set col = Nothing
    Set m_cat = Nothing

End Sub

Function adox_ExistsTable(cat, stName)
    
    Dim bFound, tbl

    bFound = False
    For Each tbl In cat.Tables
        If UCase(tbl.Name) = UCase(stName) Then
            bFound = True
            Exit For
        End If
    Next

    adox_ExistsTable = bFound

End Function
