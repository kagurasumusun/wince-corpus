DateCreated

This property indicates the date the object was created.

Syntax

None

Parameters

None

Return Values

Returns a Variant value specifying the date created. The value is Null if DateCreated is not supported by the provider.

Remarks

None

Example

The following example demonstrates how to use the DateCreated property.

Dim m_cnn, m_cat, m_tbl, m_strConn, strAct

    Set m_cnn = CreateObject("ADOCE.CONNECTION.3.1")
    Set m_cat = CreateObject("ADOXCE.CATALOG.3.1")
    Set m_tbl = CreateObject("ADOXCE.TABLE.3.1")

    m_strConn = "Provider=Microsoft.SQLSERVER.OLEDB.CE.1.0;Data Source=c:\temp\newDB.DB;"
    m_cat.Create m_strConn

    adox_CreateSmallTable m_cat, m_TEMP_TABLE, "", ""

    Set m_tbl = m_cat.Tables(m_TEMP_TABLE)
    m_cat.Tables.Refresh

    On Error Resume Next
    strAct = m_tbl.DateCreated

    Set m_cnn = Nothing
    Set m_cat = Nothing
    Set m_tbl = Nothing
End Sub

Sub adox_CreateSmallTable(cat, stName, colType, colDefault)
    Dim tbl, col
    
    Set tbl = CreateObject("ADOXCE.TABLE.3.1")
    Set col = CreateObject("ADOXCE.COLUMN.3.1")

    If colType = "" Then col.Type = adSmallInt Else col.Type = colType
    tbl.Name = stName
    col.Name = "fld1"

    tbl.Columns.Append col
    cat.Tables.Append tbl
End Sub
