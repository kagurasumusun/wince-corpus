ParentCatalog (Column)

Specifies the parent catalog of a table or column to provide access to provider-specific properties.

Syntax

None

Parameters

None

Return Values

Sets and returns a Catalog object. Setting ParentCatalog to an open Catalog allows access to provider-specific properties prior to appending a table or column to a Catalog collection.

Remarks

Some data providers allow provider-specific property values to be written only at creation (when a table or column is appended to its Catalog collection). To access these properties before appending these objects to a Catalog, specify the Catalog in the ParentCatalog property first.

An error occurs when the table or column is appended to a different Catalog than the ParentCatalog.

Example

The following code demonstrates how to append a table to a catalog.

Dim m_cat, m_col, tbl, col, strConn, stTableName, stColumnName, stExp, stAct

    Set m_cat = CreateObject("adoxce.catalog.3.1")
    Set m_col = CreateObject("adoxce.column.3.1")
    Set tbl = CreateObject("adoxce.table.3.1")
    Set col = CreateObject("adoxce.column.3.1")

    strConn = "provider=microsoft.jet.oledb.4.0;data source=temp.mdb;"
    m_cat.ActiveConnection = strConn

    stTableName = "ColAppScen18"
    If adox_ExistsTable(m_cat, stTableName)

m_cat.Tables.Delete stTableName
    End If

    tbl.Name = stTableName

    stColumnName = "Column1"
    m_col.Name = stColumnName
    m_col.Type = adSmallInt

    tbl.Columns.Append m_col
    m_cat.Tables.Append tbl

    'Create a new column
    col.Name = "Column2"
    col.Type = adSmallInt

    'Set parentcatalog
    col.ParentCatalog = m_cat

    col.Properties.Refresh

    m_cat.Tables(stTableName).Columns.Append col

    stExp = "Column2"
    stAct = m_cat.Tables(stTableName).Columns(1).Name

    'Cleanup, set all objects to Nothing.
    m_cat.Tables.Delete stTableName
    Set tbl = Nothing
    Set col = Nothing
    Set m_cat = Nothing

End Sub

Function adox_ExistsTable(cat, stName)
    
    Dim bFound, tbl

    bFound = False
    For Each tbl In cat.Tables
        If UCase(tbl.Name) = UCase(stName) Then
            bFound = True
            Exit For
        End If
    Next

    adox_ExistsTable = bFound

End Function
