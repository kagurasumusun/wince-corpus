Serial Port Driver Registry Settings

[HKEY_LOCAL_MACHINE\Drivers\BuiltIn\Serial]
    "Tsp"="Unimodem.dll" ; TAPI Service provider for the communications port
    "SysIntr"=dword:13
    "IoBase"=dword:02F8
    "IoLen"=dword:8
    "DeviceArrayIndex"=dword:0
    "Prefix"="COM"
    "Dll"="Serial.Dll"
    "Order"=dword:0
    "DeviceType"=dword:0
    "FriendlyName"=LOC_FRIENDLY_SERIAL
    "DevConfig"=hex: 10,00, 00,00, 05,00,00,00, 10,01,00,00, 
    00,4B,00,00, 00,00, 08, 00, 00, 00,00,00,00
