Code Listing for the Calc Sample

The following code example shows a simple calculation routine similar to the one found in SOAP Sample Applications, except that the following sample does not require Microsoft Foundation Classes (MFC) in the platform image. 

#include <stdio.h>
#include <mssoap.h>
LPTSTR WRAPPER_ELEMENT_NAMESPACE = _T("http://tempuri.org/message/");
#define CHK_HR(x) if(FAILED(x)) goto Done;
void Add()
{
   ISoapSerializer *pSerializer = NULL;
   ISoapReader     *pReader = NULL;
   ISoapConnector  *pConnector = NULL;
   VARIANT varEndPointURL;
   VARIANT varSoapAction;
   VARIANT varInConnectorStream;
   VARIANT varOutReaderStream;
   VARIANT_BOOL varBoolSucceeded;
   
   BSTR szResult = NULL;
   
   IStream *pInStream = NULL;
   IStream *pOutStream = NULL;
   
   IXMLDOMElement *pRPCResult = NULL;
   
   //init VARIANT's
   VariantInit(&varEndPointURL);
   VariantInit(&varSoapAction);
   VariantInit(&varInConnectorStream);
   VariantInit(&varOutReaderStream);
   
   // Connect to the service
   CHK_HR(CoCreateInstance(CLSID_HttpConnector, 
                         NULL, 
                         CLSCTX_INPROC_SERVER, 
                         IID_ISoapConnector, 
                         (void **)&pConnector));
 
   V_VT(&varEndPointURL)   = VT_BSTR;
   V_BSTR(&varEndPointURL) = SysAllocString(L"http://localhost/Calc.wsdl");
   CHK_HR(pConnector->put_Property(L"EndPointURL", varEndPointURL));
   
   // Begin message
   V_VT(&varSoapAction)   = VT_BSTR;
   V_BSTR(&varSoapAction) = SysAllocString(L"http://tempuri.org/action/Calc.Add");       
   CHK_HR(pConnector->put_Property(L"SoapAction", varSoapAction));
   CHK_HR(pConnector->BeginMessage());
   // Create the SoapSerializer
   CHK_HR(CoCreateInstance(CLSID_SoapSerializer, 
                     NULL, 
                     CLSCTX_INPROC_SERVER, 
                     IID_ISoapSerializer, 
                     (void **)&pSerializer));
   // Connect the serializer to the input stream of the connector
   CHK_HR(pConnector->get_InputStream(&pInStream));
          
   V_VT(&varInConnectorStream) = VT_UNKNOWN;
   V_UNKNOWN(&varInConnectorStream) = pInStream;
      pInStream->AddRef();  
        
   CHK_HR(pSerializer->Init(varInConnectorStream));
   // Build the SOAP Message
   CHK_HR(pSerializer->startEnvelope(L"",L"",L""));
   CHK_HR(pSerializer->startBody(L""));
   CHK_HR(pSerializer->startElement(L"Add",WRAPPER_ELEMENT_NAMESPACE,L"",L"m"));
   CHK_HR(pSerializer->startElement(L"A",L"",L"",L""));
   CHK_HR(pSerializer->writeString(L"5"));
   CHK_HR(pSerializer->endElement());
   CHK_HR(pSerializer->startElement(L"B",L"",L"",L""));
   CHK_HR(pSerializer->writeString(L"10"));
   CHK_HR(pSerializer->endElement());
   CHK_HR(pSerializer->endElement());
   CHK_HR(pSerializer->endBody());
   CHK_HR(pSerializer->endEnvelope());
   
   // Send the message to the Web service
   CHK_HR(pConnector->EndMessage());      
   // Read the response
   CHK_HR(CoCreateInstance(CLSID_SoapReader, 
                      NULL, 
                      CLSCTX_INPROC_SERVER, 
                      IID_ISoapReader, 
                      (void **)&pReader));
   // Connect the reader to the output stream of the connector
   CHK_HR(pConnector->get_OutputStream(&pOutStream));
   
   V_VT(&varOutReaderStream) = VT_UNKNOWN;
   V_UNKNOWN(&varOutReaderStream) = pOutStream;
        pOutStream->AddRef();
   
   CHK_HR(pReader->load(varOutReaderStream, L"", &varBoolSucceeded));
   
   CHK_HR(pReader->get_RPCResult(&pRPCResult));
   CHK_HR(pRPCResult->get_text(&szResult));
   
   
   // Display the result
   wprintf(L"Answer: %s\n", szResult);
   
Done:
    
   //cleanup
   VariantClear(&varEndPointURL);
   VariantClear(&varSoapAction);
   VariantClear(&varInConnectorStream);
   VariantClear(&varOutReaderStream);
   
   if(pSerializer)
      pSerializer->Release();  
   if(pReader)
      pReader->Release();
   if(pConnector)
      pConnector->Release();  
   if(pInStream)
      pInStream->Release();
   if(pOutStream)
      pOutStream->Release();
   if(pRPCResult)
      pRPCResult->Release();   
      
   if(szResult)
      SysFreeString(szResult);
}
int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance, LPWSTR lpCmdLine, int nShowCmd )
{
   CoInitializeEx(NULL, COINIT_MULTITHREADED);
   Add();
   CoUninitialize();
   return 0;
}
