IMM Registry Settings

IF LOCALE=0411
[HKEY_CURRENT_USER\Control Panel\Input Method\Hot Keys\00000030]
    "VK"=dword:19
    "KM"=dword:C001
    "Ime"=dword:e0010411

[HKEY_CURRENT_USER\Keyboard Layout\Preload\1]
    @="e0010411"

; @CESYSGEN IF CE_MODULES_MSIME98M
[HKEY_LOCAL_MACHINE\System\CurrentControlSet\Control\Layouts\e0010411]
    "Layout Text"="KBD e0010411"
    "Ime File"="msime98.dll"
; @CESYSGEN ENDIF

IF IME=PIME
; @CESYSGEN IF CE_MODULES_IMEJPP
[HKEY_LOCAL_MACHINE\System\CurrentControlSet\Control\Layouts\e0010411]
    "Layout Text"="KBD e0010411"
    "Ime File"="imejpp.dll"
; @CESYSGEN ENDIF
ENDIF

; TESTIME will override either IME98 or pIME
IF IME=TESTIME
; @CESYSGEN IF CE_MODULES_TESTIME
[HKEY_LOCAL_MACHINE\System\CurrentControlSet\Control\Layouts\e0010411]
    "Layout Text"="KBD e0010411"
    "Ime File"="testime.dll"
; @CESYSGEN ENDIF
ENDIF

ENDIF  LOCALE=0411

IF LOCALE=0412
; @CESYSGEN IF CE_MODULES_MSHIME97

    "VK"=dword:19
    "KM"=dword:0400
    "Ime"=dword:e0010412

[HKEY_CURRENT_USER\Control Panel\Input Method\Hot Keys\00000052]
    "VK"=dword:15
    "KM"=dword:0400
    "Ime"=dword:e0010412

[HKEY_CURRENT_USER\Keyboard Layout\Preload\1]

    "Ime File"="mshime97.dll"

[HKEY_LOCAL_MACHINE\System\Ime]
    "Settings"=dword:16
; @CESYSGEN ENDIF
ENDIF LOCALE=0412
