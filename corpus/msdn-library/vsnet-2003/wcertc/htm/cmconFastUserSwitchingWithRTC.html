Fast User Switching with the Communications Service

The following code example shows the behavior of a Communications Service application when a fast user switch (FUS) occurs. The application should shut down all active sessions when the FUS is detected. The operations in the Initialize the Communications Service example must be performed before using this code.

Note   This code has not been thoroughly tested, does not contain error checking, and is not intended for production use.
//
// Code 1 - Register to receive terminal notification events
//
#include <wtsapi32.h>
typedef BOOL (WINAPI *WTSREGISTERSESSIONNOTIFICATION)(HWND, DWORD);
typedef BOOL (WINAPI *WTSUNREGISTERSESSIONNOTIFICATION)(HWND);
//
// register for terminal services notifications
//
HWND     hWND
HANDLE   hWtsLib;
hWtsLib = LoadLibrary( _T("wtsapi32.dll") );
if (hWtsLib)
{
    WTSREGISTERSESSIONNOTIFICATION   fnWtsRegisterSessionNotification;
    
    fnWtsRegisterSessionNotification = 
            (WTSREGISTERSESSIONNOTIFICATION)GetProcAddress( hWtsLib, 
                                  "WTSRegisterSessionNotification" );
    if (fnWtsRegisterSessionNotification)
    {
        fnWtsRegisterSessionNotification( hWnd, 
                                          NOTIFY_FOR_THIS_SESSION );
    }
}
//
// Code 2 - Event handler for the Windows Terminal Services (WTS)
//             Session Events
//
// On WM_WTSSESSION_CHANGE
{
HRESULT hr;
switch( wParam )
{
    case WTS_CONSOLE_CONNECT:
        // Listen for incoming calls
        hr = pRTCClient->put_ListenForIncomingSessions( RTCLM_BOTH );

        // if(FAILED(hr))  insert error code here
        break;
    case WTS_CONSOLE_DISCONNECT:
        // If a call is active, terminate it
        if (pSession)
        {
            hr = pSession->Terminate(RTCTR_SHUTDOWN);
            pSession->Release();
            pSession = NULL;

            // if (FAILED(hr))  insert error code here
        }
        // Do not listen for incoming calls 
        hr = pRTCClient->put_ListenForIncomingSessions(RTCLM_NONE);

        if(FAILED(hr))  insert error code here
        break;
}
}
//
// Code 3 - Unregister for terminal services notifications
//
if (hWtsLib)
{
    WTSUNREGISTERSESSIONNOTIFICATION 
                                   fnWtsUnRegisterSessionNotification;
    fnWtsUnRegisterSessionNotification = 
           (WTSUNREGISTERSESSIONNOTIFICATION)GetProcAddress(hWtsLib,
                                 "WTSUnRegisterSessionNotification" );
        if (fnWtsUnRegisterSessionNotification)
        {
            fnWtsUnRegisterSessionNotification( hWnd );
        }
        FreeLibrary( hWtsLib );
        hWtsLib = NULL;
    }
