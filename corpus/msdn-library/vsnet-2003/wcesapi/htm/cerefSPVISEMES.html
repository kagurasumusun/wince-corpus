SPVISEMES

This enumeration defines the SAPI 5 Viseme set. This set is based on the Disney 13 Visemes. Examples given are for the SAPI 5 English Phoneme set.

typedef enum SPVISEMES{
// English examples
//------------------
SP_VISEME_0 = 0,// silence
SP_VISEME_1,// ae, ax, ah
SP_VISEME_2,// aa
SP_VISEME_3,// ao
SP_VISEME_4,// ey, eh, uh
SP_VISEME_5,// er
SP_VISEME_6,// y, iy, ih, ix
SP_VISEME_7,// w, uw
SP_VISEME_8,// ow
SP_VISEME_9,// aw
SP_VISEME_10,  // oy
SP_VISEME_11,  // ay
SP_VISEME_12,  // h
SP_VISEME_13,  // r
SP_VISEME_14,  // l
SP_VISEME_15,  // s, z
SP_VISEME_16,  // sh, ch, jh, zh
SP_VISEME_17,  // th, dh
SP_VISEME_18,  // f, v
SP_VISEME_19,  // d, t, n
SP_VISEME_20,  // k, g, ng
SP_VISEME_21,  // p, b, m
} SPVISEMES;
