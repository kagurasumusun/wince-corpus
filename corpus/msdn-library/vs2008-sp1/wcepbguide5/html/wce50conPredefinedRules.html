Rule Command Default action

.cpp.obj $(CPP) $(CPPFLAGS) /c $*.cpp cl /c $*.cpp
.cxx.exe $(CXX) $(CXXFLAGS) $*.cxx cl $*.cxx

.rc.res $(RC) $(RFLAGS) /r $* rc /r $*
