APPLICATION E-Mail Configuration Service Provider for POP3 Example

Send Feedback on this topic to the authors

The following example shows how to use the APPLICATION Configuration Service Provider to create a POP3 account. 

<characteristic type="APPLICATION">
  <parm name="APPID" value="110"/>
  <parm name="PROVIDER-ID" value="MyMail"/>
  <parm name="TO-NAPID" value="IAP1"/>
  <characteristic type="APPADDR">
    <parm name="ADDR" value="pop.mail.com"/>
    <characteristic type="PORT">
      <parm name="PORTNBR" value="110"/>
      <parm name="SERVICE" value="STARTTLS"/>
    </characteristic>
  </characteristic>
  <characteristic type="APPAUTH">
    <parm name="AAUTHTYPE" value="CRAM-MD5"/>
    <parm name="AAUTHNAME" value="username"/>
    <parm name="AAUTHSECRET" value="password"/>
  </characteristic>
</characteristic>
 
<characteristic type="APPLICATION">
  <parm name="APPID" value="110"/>
  <parm name="TO-NAPID" value="IAP1"/>
  <parm name="ADDR" value="pop.mail.com"/> 
</characteristic>
