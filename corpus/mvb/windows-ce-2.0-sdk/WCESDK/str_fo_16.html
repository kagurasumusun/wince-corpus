LV_ITEM

The LV_ITEM structure specifies or receives the attributes of a list view item.

Syntax

typedef struct _LV_ITEM {
 UINT mask;
int iItem;
int iSubItem;
UINT state;
UINT stateMask;
LPSTR pszText;
int cchTextMax;
int iImage;
LPARAM lParam;
} LV_ITEM;
