COAUTHINFO

The COAUTHINFO structure is used to determine the authentication settings.

Syntax

typedef struct _COAUTHINFO {
DWORD dwAuthnSvc;
DWORD dwAuthzSvc;
[string] WCHAR * pwszServerPrincName;
DWORD dwAuthnLevel;
DWORD ddwImpersonationLevel;
AUTH_IDENTITY *pAuthIdentityData;
DWORD dwCapabilities;
} COAUTHINFO;
