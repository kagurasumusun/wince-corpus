NM_TREEVIEW

The NM_TREEVIEW structure contains information about a tree view notification message.

typedef struct _NM_TREEVIEW { nmtv
 NMHDR hdr;
UINT action;
TV_ITEM itemOld;

} NM_TREEVIEW;
