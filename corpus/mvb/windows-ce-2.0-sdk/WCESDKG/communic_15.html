Sample Infrared Socket Server

This simple IrSock server allocates a socket and binds it to the service named "MyServer". It then allocates a single connection object and prepares the server to listen for incoming connections. When the client contacts the server, the server accepts the connection. It then receives a string from the client, passes one back, and closes the connection.

#include <windows.h>
#include <af_irda.h>

int WINAPI WinMain(HINSTANCE hInstance, HINSTANCE hPrevInstance,
                    LPTSTR lpCmdLine, int nCmdShow )
{
    SOCKET         ServerSock, // Socket for the Server
                   ClientSock; // Socket for the Client

    // The IrDA address for the server
    SOCKADDR_IRDA  address = {AF_IRDA, 0, 0, 0, 0, "MyServer"};

    char           helloServer[25];  // ASCII String 
    TCHAR          helloText[25];    // UNICODE String
    int            idx = 0;

    // Allocate a stream socket in the IrDA address domain
    ServerSock = socket(AF_IRDA, SOCK_STREAM, 0);

    // Associate the service name MyServer with the socket
    bind(ServerSock, (struct sockaddr *)&address, sizeof(address));

    // Allow a single incoming connection
    listen(ServerSock, 1);

    // Wait for a client to contact us.  Blocking call.
    ClientSock = accept(ServerSock, 0, 0);

    // Receive a string from the client
    recv(ClientSock, helloServer, sizeof(helloServer), 0);

    // Convert string from client to Unicode
    for (idx = 0; idx <= sizeof(helloServer); idx++)
          helloText[idx] = helloServer[idx];

    // Display string received from client
    MessageBox (NULL, helloText, TEXT("IR Server"), MB_OK);

    // Send a greeting back to the Client
    send(ClientSock, "Hello Client!", strlen("Hello Client!")+1, 0);

    closesocket(ClientSock); // Close Client Socket
    closesocket(ServerSock); // Close Server Socket

return 0;
}
